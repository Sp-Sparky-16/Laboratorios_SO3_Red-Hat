PRACTICA 1: CIFRADO CON GPG2
-----------------------------------------------------------------------------

# 1. Instalar gpg2 (ejecutar según la distribución)
# En Debian/Ubuntu:
sudo apt update && sudo apt install gnupg2 -y
# En RHEL/Rocky Linux:
sudo dnf install gnupg2 -y

# 2. Crear un directorio y un archivo de prueba
mkdir ~/practica_cifrado
cd ~/practica_cifrado
echo "Este es el contenido confidencial del archivo de seguridad" > documento.txt

# 3. Cifrar el archivo de forma simétrica (solicitará una contraseña en pantalla)
gpg2 --symmetric --cipher-algo AES256 documento.txt

# 4. Intentar acceder al archivo cifrado (mostrará caracteres ilegibles binarios)
cat documento.txt.gpg

# 5. Descifrar el archivo mostrando su contenido por consola
gpg2 --decrypt documento.txt.gpg

PRACTICA 2: CONTROL DE TRAFICO CON IPTABLES Y UFW / FIREWALL-CMD
-----------------------------------------------------------------------------

### FASE 1: HABILITAR SERVICIOS (HTTP, FTP, SSH) ###

# En Debian/Ubuntu:
sudo apt install apache2 vsftpd openssh-server -y
sudo systemctl enable --now apache2 vsftpd ssh

# En RHEL/Rocky Linux:
sudo dnf install httpd vsftpd openssh-server -y
sudo systemctl enable --now httpd vsftpd sshd

# Comprobar el estado de los servicios:
sudo systemctl status apache2 httpd vsftpd ssh sshd --no-pager


### FASE 2: BLOQUEO CON IPTABLES ###

# Bloquear tráfico entrante en los puertos 80 (HTTP), 21 (FTP) y 22 (SSH)
sudo iptables -A INPUT -p tcp --dport 80 -j DROP
sudo iptables -A INPUT -p tcp --dport 21 -j DROP
sudo iptables -A INPUT -p tcp --dport 22 -j DROP

# Mostrar las reglas activas de iptables para verificar el bloqueo
sudo iptables -L -n -v


### FASE 3: DESBLOQUEO CON IPTABLES ###

# Eliminar las reglas de bloqueo previamente añadidas
sudo iptables -D INPUT -p tcp --dport 80 -j DROP
sudo iptables -D INPUT -p tcp --dport 21 -j DROP
sudo iptables -D INPUT -p tcp --dport 22 -j DROP


### FASE 4: MANEJO CON EL FIREWALL DE LA DISTRIBUCIÓN ###

# ---- OPCIÓN A: Si tu distro es basada en DEBIAN (UFW) ----
# Habilitar el firewall
sudo ufw enable

# Bloquear los puertos
sudo ufw deny 80/tcp
sudo ufw deny 21/tcp
sudo ufw deny 22/tcp
sudo ufw status verbose

# Permitir y restablecer el tráfico de nuevo
sudo ufw allow 80/tcp
sudo ufw allow 21/tcp
sudo ufw allow 22/tcp
sudo ufw status verbose


# ---- OPCIÓN B: Si tu distro es basada en RED HAT (Firewall-cmd) ----
# Bloquear los puertos
sudo firewall-cmd --permanent --remove-service={http,ftp,ssh}
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="80" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="21" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="22" protocol="tcp" drop'
sudo firewall-cmd --reload
sudo firewall-cmd --list-all

# Permitir y restablecer el tráfico de nuevo (Remover reglas rich y habilitar servicios estándar)
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="80" protocol="tcp" drop'
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="21" protocol="tcp" drop'
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="22" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-service={http,ftp,ssh}
sudo firewall-cmd --reload
sudo firewall-cmd --list-all

PRACTICA 3: INSTALACION Y CONFIGURACION DE IDS SNORT
-----------------------------------------------------------------------------

# 1. Instalar Snort
# En Debian/Ubuntu:
sudo apt update && sudo apt install snort -y
# En RHEL/Rocky Linux:
sudo dnf install snort -y

# 2. Agregar las reglas personalizadas al archivo local.rules
# Abrir el archivo de reglas locales:
# (Para sistemas Debian suele ubicarse en /etc/snort/rules/local.rules)
# (Para compilaciones o RHEL verificar ruta, comúnmente /etc/snort/local.rules)

sudo bash -c 'cat >> /etc/snort/rules/local.rules << EOF
alert icmp any any -> any any (msg:"ALERTA: Trafico ICMP detectado (Ping)"; sid:1000001; rev:1;)
alert tcp any any -> any 21 (msg:"ALERTA: Trafico de conexion al puerto 21 FTP"; sid:1000002; rev:1;)
alert tcp any any -> any 22 (msg:"ALERTA: Trafico de conexion al puerto 22 SSH"; sid:1000003; rev:1;)
alert tcp any any -> any 80 (msg:"ALERTA: Trafico de conexion al puerto 80 HTTP"; sid:1000004; rev:1;)
EOF'

# 3. Validar la configuración de Snort para asegurar que no contenga errores
sudo snort -T -c /etc/snort/snort.conf

# 4. Ejecutar el IDS Snort en modo escucha apuntando a la interfaz de red activa
# NOTA: Reemplazar "eth0" o "ens33" por el nombre real de tu interfaz (verificar con 'ip a')
sudo snort -A console -q -c /etc/snort/snort.conf -i eth0

PRACTICA 4: CONFIGURAR 2FA CON GOOGLE AUTHENTICATOR (PAM) PARA SSH
-----------------------------------------------------------------------------

# 1. Instalar el módulo PAM de Google Authenticator
# En Debian/Ubuntu:
sudo apt update && sudo apt install libpam-google-authenticator -y
# En RHEL/Rocky Linux (Requiere repositorio EPEL previo):
sudo dnf install epel-release -y && sudo dnf install google-authenticator -y

# 2. Ejecutar la configuración de doble factor con el usuario del laboratorio
# IMPORTANTE: Responder 'y' (yes) a las preguntas principales y escanear el código QR con el móvil.
google-authenticator

# 3. Configurar la integración del módulo PAM
# Editar el archivo de configuración PAM para SSH:
sudo nano /etc/pam.d/sshd
# Añadir la siguiente línea al final o al inicio del archivo según la directiva del sistema:
auth required pam_google_authenticator.so nullok

# 4. Configurar el demonio SSH para admitir la autenticación multifactor
sudo nano /etc/ssh/sshd_config
# Modificar o verificar que los siguientes parámetros estén configurados correctamente:
KbdInteractiveAuthentication yes
ChallengeResponseAuthentication yes
UsePAM yes

# Si deseas requerir explícitamente tanto la clave pública/contraseña MÁS el código 2FA:
# AuthenticationMethods password,keyboard-interactive

# 5. Reiniciar el servicio SSH para aplicar los cambios establecidos
# En Debian/Ubuntu:
sudo systemctl restart ssh
# En RHEL/Rocky Linux:
sudo systemctl restart sshd

# 6. Comando de prueba desde la máquina Host externa:
# ssh tu_usuario@IP_DE_LA_VM
