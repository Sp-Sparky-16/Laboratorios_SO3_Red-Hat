====================
Practica 1 - Cifrado
====================
# Instalar el paquete en RHEL 10
sudo dnf install -y gnupg2

# Verificar que la herramienta esté disponible
gpg2 --version

# Crear el directorio de la práctica y acceder a él
mkdir -p ~/practica_cifrado
cd ~/practica_cifrado
echo "Este es el contenido confidencial del archivo de seguridad" > documento.txt

# 3. Cifrar el archivo de forma simétrica (solicitará una contraseña en pantalla)
gpg2 --symmetric --cipher-algo AES256 documento.txt
# Crear un archivo de prueba con texto dentro
echo "Este es el contenido confidencial del archivo." > documento.txt

# 4. Intentar acceder al archivo cifrado (mostrará caracteres ilegibles binarios)
cat documento.txt.gpg
# Verificar el contenido original
cat documento.txt

# 5. Descifrar el archivo mostrando su contenido por consola
gpg2 --decrypt documento.txt.gpg
gpg2 --pinentry-mode loopback -c documento.txt

# Intentar leer el archivo cifrado con cat
cat documento.txt.gpg

### FASE 2: BLOQUEO CON IPTABLES ###
# Asegurar la instalación de la herramienta iptables
sudo dnf install -y iptables iptables-services

# Bloquear tráfico entrante en los puertos 80 (HTTP), 21 (FTP) y 22 (SSH)
# Bloquear las conexiones entrantes a los puertos 80 (HTTP), 21 (FTP) y 22 (SSH)
sudo iptables -A INPUT -p tcp --dport 80 -j DROP
sudo iptables -A INPUT -p tcp --dport 21 -j DROP
sudo iptables -A INPUT -p tcp --dport 22 -j DROP

# Mostrar las reglas activas de iptables para verificar el bloqueo
# Verificar que las reglas estén aplicadas

# Eliminar las reglas de bloqueo agregadas previamente

sudo iptables -D INPUT -p tcp --dport 21 -j DROP
sudo iptables -D INPUT -p tcp --dport 22 -j DROP

### FASE 4: MANEJO CON EL FIREWALL DE LA DISTRIBUCIÓN ###

# ---- OPCIÓN A: Si tu distro es basada en DEBIAN (UFW) ----
# Habilitar el firewall
sudo ufw enable
# Asegurar que firewalld esté iniciado
sudo systemctl enable --now firewalld

# Bloquear los puertos
sudo ufw deny 80/tcp
sudo ufw deny 21/tcp
sudo ufw deny 22/tcp
sudo ufw status verbose
# Bloquear los puertos agregando la regla de rechazo (rich-rules o remoción de servicios)
sudo firewall-cmd --add-rich-rule='rule family="ipv4" port port="80" protocol="tcp" reject'
sudo firewall-cmd --add-rich-rule='rule family="ipv4" port port="21" protocol="tcp" reject'
sudo firewall-cmd --add-rich-rule='rule family="ipv4" port port="22" protocol="tcp" reject'

# Permitir y restablecer el tráfico de nuevo
sudo ufw allow 80/tcp
sudo ufw allow 21/tcp
sudo ufw allow 22/tcp
sudo ufw status verbose
# Listar la configuración activa para verificar los bloqueos
sudo firewall-cmd --list-all

# Retirar las reglas de bloqueo
sudo firewall-cmd --remove-rich-rule='rule family="ipv4" port port="80" protocol="tcp" reject'
sudo firewall-cmd --remove-rich-rule='rule family="ipv4" port port="21" protocol="tcp" reject'
sudo firewall-cmd --remove-rich-rule='rule family="ipv4" port port="22" protocol="tcp" reject'

# ---- OPCIÓN B: Si tu distro es basada en RED HAT (Firewall-cmd) ----
# Bloquear los puertos
sudo firewall-cmd --permanent --remove-service={http,ftp,ssh}
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="80" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="21" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-rich-rule='rule family="ipv4" port port="22" protocol="tcp" drop'
# Permitir los servicios de forma explícita
sudo firewall-cmd --add-service={http,ftp,ssh} --permanent
sudo firewall-cmd --reload
sudo firewall-cmd --list-all

# Permitir y restablecer el tráfico de nuevo (Remover reglas rich y habilitar servicios estándar)
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="80" protocol="tcp" drop'
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="21" protocol="tcp" drop'
sudo firewall-cmd --permanent --remove-rich-rule='rule family="ipv4" port port="22" protocol="tcp" drop'
sudo firewall-cmd --permanent --add-service={http,ftp,ssh}
sudo firewall-cmd --reload
# Confirmar la apertura de puertos en la zona activa
sudo firewall-cmd --list-all

# En Debian/Ubuntu:

# 2. Agregar las reglas personalizadas al archivo local.rules
# Abrir el archivo de reglas locales:
# (Para sistemas Debian suele ubicarse en /etc/snort/rules/local.rules)
# (Para compilaciones o RHEL verificar ruta, comúnmente /etc/snort/local.rules)
find /usr -name "libpcre2-8.pc" 2>/dev/null

sudo bash -c 'cat >> /etc/snort/rules/local.rules << EOF
alert icmp any any -> any any (msg:"ALERTA: Trafico ICMP detectado (Ping)"; sid:1000001; rev:1;)
alert tcp any any -> any 21 (msg:"ALERTA: Trafico de conexion al puerto 21 FTP"; sid:1000002; rev:1;)
alert tcp any any -> any 22 (msg:"ALERTA: Trafico de conexion al puerto 22 SSH"; sid:1000003; rev:1;)
alert tcp any any -> any 80 (msg:"ALERTA: Trafico de conexion al puerto 80 HTTP"; sid:1000004; rev:1;)
EOF'
export PKG_CONFIG_PATH=/usr/lib64/pkgconfig:/usr/share/pkgconfig:$PKG_CONFIG_PATH

# 3. Validar la configuración de Snort para asegurar que no contenga errores
sudo snort -T -c /etc/snort/snort.conf
cd ~
git clone https://github.com/PCRE2Project/pcre2.git
